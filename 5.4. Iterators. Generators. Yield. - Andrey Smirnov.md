```python
class FlatIterator:
    def __init__(self, list_of_list):
        self.list_of_list = list_of_list
        self.outer_index = 0
        self.inner_index = 0

    def __iter__(self):
        return self

    def __next__(self):
        # Если все списки исчерпаны – стоп
        if self.outer_index >= len(self.list_of_list):
            raise StopIteration

        # Если внутренний индекс вышел за границы текущего списка – переходим к следующему
        while self.outer_index < len(self.list_of_list) and self.inner_index >= len(self.list_of_list[self.outer_index]):
            self.outer_index += 1
            self.inner_index = 0

        # Если после переключения все списки закончились
        if self.outer_index >= len(self.list_of_list):
            raise StopIteration

        # Берём текущий элемент, увеличиваем внутренний индекс и возвращаем его
        item = self.list_of_list[self.outer_index][self.inner_index]
        self.inner_index += 1
        return item


def test_1():
    list_of_lists_1 = [
        ['a', 'b', 'c'],
        ['d', 'e', 'f', 'h', False],
        [1, 2, None]
    ]

    for flat_iterator_item, check_item in zip(
            FlatIterator(list_of_lists_1),
            ['a', 'b', 'c', 'd', 'e', 'f', 'h', False, 1, 2, None]
    ):
        assert flat_iterator_item == check_item

    assert list(FlatIterator(list_of_lists_1)) == ['a', 'b', 'c', 'd', 'e', 'f', 'h', False, 1, 2, None]


if __name__ == '__main__':
    test_1()
```

```python
class FlatIterator:
    def __init__(self, list_of_list):
        self.list_of_list = list_of_list
        # Стек для пар [список, текущий индекс]
        self.stack = [[list_of_list, 0]]

    def __iter__(self):
        return self

    def __next__(self):
        while self.stack:
            current_list, index = self.stack[-1]
            if index >= len(current_list):
                self.stack.pop()
                continue
            item = current_list[index]
            self.stack[-1][1] += 1
            if isinstance(item, list):
                self.stack.append([item, 0])
                continue
            return item
        raise StopIteration


def test_3():
    list_of_lists_2 = [
        [['a'], ['b', 'c']],
        ['d', 'e', [['f'], 'h'], False],
        [1, 2, None, [[[[['!']]]]], []]
    ]

    for flat_iterator_item, check_item in zip(
            FlatIterator(list_of_lists_2),
            ['a', 'b', 'c', 'd', 'e', 'f', 'h', False, 1, 2, None, '!']
    ):
        assert flat_iterator_item == check_item

    assert list(FlatIterator(list_of_lists_2)) == ['a', 'b', 'c', 'd', 'e', 'f', 'h', False, 1, 2, None, '!']
```

```python
import types

def flat_generator(list_of_list):
    for item in list_of_list:
        if isinstance(item, list):
            yield from flat_generator(item)
        else:
            yield item
```
